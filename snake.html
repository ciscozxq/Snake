            background-color: #11111b;

            border: 3px solid #89b4fa;

            box-shadow: 0 4px 15px rgba(0, 0, 0, 0.5);

        }

<body>


    <div id="score-board">得分: <span id="score">0</span></div>

    <canvas id="gameCanvas" width="400" height="400"></canvas>

    <div class="controls-hint">使用 <b>方向键 (↑ ↓ ← →)</b> 或 <b>W A S D</b> 控制移动 | 按 <b>空格键</b> 重新开始</div>


    <script>

        const canvas = document.getElementById("gameCanvas");

        const ctx = canvas.getContext("2d");

        const scoreElement = document.getElementById("score");


        const gridSize = 20; // 每个格子的像素大小

        const tileCount = canvas.width / gridSize; // 网格数量 (20x20)


        let snake = [];

        let food = {};

        let dx = 0;

        let dy = 0;

        let score = 0;

        let gameLoop = null;

        let isGameOver = false;

        let changingDirection = false; // 防止单帧内多次按键导致反向自杀


        // 初始化游戏

        function initGame() {

            snake = [

                { x: 10, y: 10 },

                { x: 9, y: 10 },

                { x: 8, y: 10 }

            ];

            dx = 1; // 初始向右移动

            dy = 0;

            score = 0;

            isGameOver = false;

            scoreElement.innerText = score;

            spawnFood();


            if (gameLoop) clearInterval(gameLoop);

            gameLoop = setInterval(update, 100); // 100ms 刷新一帧

        }


        // 游戏主循环

        function update() {

            if (isGameOver) return;

            changingDirection = false;


            moveSnake();

            if (checkCollision()) {

                handleGameOver();

                return;

            }

            draw();

        }


        // 移动蛇的逻辑

        function moveSnake() {

            const head = { x: snake[0].x + dx, y: snake[0].y + dy };

            snake.unshift(head); // 插入新头部


            // 判断是否吃到食物

            if (head.x === food.x && head.y === food.y) {

                score += 10;

                scoreElement.innerText = score;

                spawnFood();

            } else {

                snake.pop(); // 没吃到则移除尾部

            }

        }


        // 碰撞检测（撞墙或撞自己）

        function checkCollision() {

            const head = snake[0];

            // 撞墙

            if (head.x < 0 || head.x >= tileCount || head.y < 0 || head.y >= tileCount) {

                return true;

            }

            // 撞身体

            for (let i = 1; i < snake.length; i++) {

                if (head.x === snake[i].x && head.y === snake[i].y) {

                    return true;

                }

            }

            return false;

        }


        // 随机生成食物（避开蛇身）

        function spawnFood() {

            food = {

                x: Math.floor(Math.random() * tileCount),

                y: Math.floor(Math.random() * tileCount)

            };

            for (let part of snake) {

                if (part.x === food.x && part.y === food.y) {

                    spawnFood();

                    break;

                }

            }

        }


        // 渲染画面

        function draw() {

            // 清空画布

            ctx.fillStyle = "#11111b";

            ctx.fillRect(0, 0, canvas.width, canvas.height);


            // 画食物

            ctx.fillStyle = "#f38ba8";

            ctx.beginPath();

            ctx.arc(food.x * gridSize + gridSize / 2, food.y * gridSize + gridSize / 2, gridSize / 2 - 2, 0, Math.PI * 2);

            ctx.fill();


            // 画蛇

            snake.forEach((part, index) => {

                ctx.fillStyle = index === 0 ? "#a6e3a1" : "#94e2d5"; // 蛇头和蛇身不同颜色

                ctx.fillRect(part.x * gridSize + 1, part.y * gridSize + 1, gridSize - 2, gridSize - 2);

            });

        }


        // 游戏结束处理

        function handleGameOver() {

            isGameOver = true;

            clearInterval(gameLoop);

            ctx.fillStyle = "rgba(0, 0, 0, 0.7)";

            ctx.fillRect(0, 0, canvas.width, canvas.height);


            ctx.fillStyle = "#f38ba8";

            ctx.font = "bold 36px sans-serif";

            ctx.textAlign = "center";

            ctx.fillText("游戏结束!", canvas.width / 2, canvas.height / 2 - 10);


            ctx.fillStyle = "#cdd6f4";

            ctx.font = "16px sans-serif";

            ctx.fillText("按空格键重新开始", canvas.width / 2, canvas.height / 2 + 25);

        }


        // 键盘事件监听

        document.addEventListener("keydown", (e) => {

            if (e.code === "Space" && isGameOver) {

                initGame();

                return;

            }


            if (changingDirection) return;


            const key = e.key.toLowerCase();

            const goingUp = dy === -1;

            const goingDown = dy === 1;

            const goingRight = dx === 1;

            const goingLeft = dx === -1;


            if ((key === "arrowleft" || key === "a") && !goingRight) {

                dx = -1; dy = 0; changingDirection = true;

            } else if ((key === "arrowup" || key === "w") && !goingDown) {

                dx = 0; dy = -1; changingDirection = true;

            } else if ((key === "arrowright" || key === "d") && !goingLeft) {

                dx = 1; dy = 0; changingDirection = true;

            } else if ((key === "arrowdown" || key === "s") && !goingUp) {

                dx = 0; dy = 1; changingDirection = true;

            }

        });


        // 启动游戏

        initGame();

    </script>
